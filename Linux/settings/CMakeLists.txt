pkg_check_modules(ADW IMPORTED_TARGET libadwaita-1 gtk4 fontconfig)
if(ADW_FOUND)
  add_executable(mikey-settings Settings.cpp Guide.cpp Emoji.cpp ../ibus/Prefs.cpp)
  target_compile_definitions(mikey-settings PRIVATE VIKEY_VERSION="${VIKEY_VERSION}" VIKEY_DATADIR="${CMAKE_INSTALL_FULL_DATADIR}/mikey")
  target_link_libraries(mikey-settings PRIVATE PkgConfig::ADW)
  pkg_check_modules(GTKX11 IMPORTED_TARGET gtk4-x11 x11)
  if(GTKX11_FOUND)   # the emoji picker places itself in the middle of the screen (X11; Wayland has no window positions)
    target_compile_definitions(mikey-settings PRIVATE VIKEY_HAVE_X11)
    target_link_libraries(mikey-settings PRIVATE PkgConfig::GTKX11)
  endif()
  install(TARGETS mikey-settings DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES mikey.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES ${ROOT}/site/assets/icon.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME mikey.png)
else()
  message(WARNING "libadwaita not found: mikey-settings not built")
endif()
